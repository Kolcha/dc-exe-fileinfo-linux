cmake_minimum_required(VERSION 3.20)
project(file_info VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This plugin supports Linux only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Qt6 6.7 REQUIRED COMPONENTS Widgets)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBELF REQUIRED IMPORTED_TARGET libelf)
add_library(binary_analysis STATIC src/analysis/ElfAnalyzer.cpp src/analysis/DependencyResolver.cpp)
target_include_directories(binary_analysis PUBLIC src)
target_link_libraries(binary_analysis PRIVATE PkgConfig::LIBELF)
set_target_properties(binary_analysis PROPERTIES CXX_VISIBILITY_PRESET hidden)
add_library(file_info_ui STATIC src/ui/ViewerWidget.cpp src/ui/DependenciesModel.cpp)
target_link_libraries(file_info_ui PUBLIC binary_analysis Qt6::Widgets)
set_target_properties(file_info_ui PROPERTIES CXX_VISIBILITY_PRESET hidden)
add_library(file_info_qt6 MODULE src/wlx/plugin.cpp)
target_include_directories(file_info_qt6 PRIVATE sdk)
target_link_libraries(file_info_qt6 PRIVATE file_info_ui)
target_link_options(file_info_qt6 PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/src/wlx/exports.map")
set_property(TARGET file_info_qt6 APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/wlx/exports.map")
set_target_properties(file_info_qt6 PROPERTIES PREFIX "" SUFFIX ".wlx" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
foreach(target binary_analysis file_info_ui file_info_qt6)
  set_target_properties(${target} PROPERTIES VISIBILITY_INLINES_HIDDEN YES)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
option(BUILD_DEMO_VIEWER "Build standalone viewer for manual inspection" OFF)
if(BUILD_DEMO_VIEWER)
  add_executable(file_info src/ui/main.cpp)
  target_link_libraries(file_info PRIVATE file_info_ui)
endif()
install(TARGETS file_info_qt6 LIBRARY DESTINATION lib/doublecmd/plugins/wlx/file_info)
