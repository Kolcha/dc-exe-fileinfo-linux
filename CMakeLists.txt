cmake_minimum_required(VERSION 3.20)
project(exe_file_info VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(USE_QT_MAJOR_VERSION "6" CACHE STRING "Major Qt version to use")
find_package(Qt${USE_QT_MAJOR_VERSION} REQUIRED COMPONENTS Widgets)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBELF REQUIRED IMPORTED_TARGET libelf)
add_library(binary_analysis STATIC src/analysis/ElfAnalyzer.cpp src/analysis/DependencyResolver.cpp)
target_include_directories(binary_analysis PUBLIC src)
target_link_libraries(binary_analysis PRIVATE PkgConfig::LIBELF)
set_target_properties(binary_analysis PROPERTIES CXX_VISIBILITY_PRESET hidden)
add_library(file_info_ui STATIC src/ui/ViewerWidget.cpp src/ui/DependenciesModel.cpp)
target_link_libraries(file_info_ui PUBLIC binary_analysis Qt::Widgets)
set_target_properties(file_info_ui PROPERTIES CXX_VISIBILITY_PRESET hidden)
set(plugin_target exe_file_info_qt${USE_QT_MAJOR_VERSION})
add_library(${plugin_target} MODULE src/wlx/plugin.cpp)
target_include_directories(${plugin_target} PRIVATE sdk)
target_link_libraries(${plugin_target} PRIVATE file_info_ui)
target_link_options(${plugin_target} PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/src/wlx/exports.map")
set_property(TARGET ${plugin_target} APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/wlx/exports.map")
set_target_properties(${plugin_target} PROPERTIES PREFIX "" SUFFIX ".wlx" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
foreach(target binary_analysis file_info_ui ${plugin_target})
  set_target_properties(${target} PROPERTIES VISIBILITY_INLINES_HIDDEN YES)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
option(BUILD_DEMO_VIEWER "Build standalone viewer for manual inspection" OFF)
if(BUILD_DEMO_VIEWER)
  add_executable(exe_file_info src/ui/main.cpp)
  target_link_libraries(exe_file_info PRIVATE file_info_ui)
endif()
install(TARGETS ${plugin_target} LIBRARY DESTINATION lib/doublecmd/plugins/wlx/exe_file_info)
